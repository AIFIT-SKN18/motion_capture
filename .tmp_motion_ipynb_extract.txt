DEFAULT_SETUP   = 1   # S

MAX_FRAMES = None        # None이면 전체, 숫자면 그만큼만(디버그)

# --------- 추가: 타깃 FPS / 길이 / 인원 ---------

# --------- MediaPipe → NTU25 매핑 ---------
def mp_to_ntu25(landmarks):
    """ landmarks: list of 33 normalized (x,y,visibility) in [0,1]
        return (25,3) with [x,y,score] in [0,1]
    """
    def get(i):
        lm = landmarks[i]
        return np.array([lm.x, lm.y, getattr(lm, "visibility", 1.0)], dtype=np.float32)
    def mid(a,b): return (a+b)/2.0

    # key indices
    LSH, RSH = get(11), get(12)
    LHP, RHP = get(23), get(24)
    L_EL, R_EL = get(13), get(14)
    L_WR, R_WR = get(15), get(16)
    L_ID, R_ID = get(19), get(20)
    L_TB, R_TB = get(21), get(22)
    L_KN, R_KN = get(25), get(26)
    L_AN, R_AN = get(27), get(28)
    L_FI, R_FI = get(31), get(32)
    SHO = mid(LSH, RSH)  # spine shoulder
    HIP = mid(LHP, RHP)  # spine base
    NEK = SHO            # neck approx
    HED = get(0)         # head ~ nose

    J = np.stack([
        HIP,           # 0 spine base
        (HIP+SHO)/2.0, # 1 spine mid (approx)
        NEK,           # 2 neck
        HED,           # 3 head
        LSH, L_EL, L_WR, L_ID,       # 4..7
        RSH, R_EL, R_WR, R_ID,       # 8..11
        LHP, L_KN, L_AN, L_FI,       # 12..15
        RHP, R_KN, R_AN, R_FI,       # 16..19
        SHO, L_ID, L_TB, R_ID, R_TB  # 20..24
    ], axis=0).astype(np.float32)

    J[:, :2] = np.clip(J[:, :2], 0.0, 1.0)
    J[:, 2]  = np.clip(J[:, 2], 0.0, 1.0)  # score
    return J  # (25,3)

# --------- .skeleton 파일 작성기 ---------
def write_ntu_skeleton(out_path, frames_joint):
    """
    frames_joint: list of dict per frame:
      - if no body: {'numBody': 0}
      - if body:    {'numBody': 1, 'J': (25,3) normalized}
    Writes NTU skeleton text format compatible with read_skeleton_filter.
    """
    with open(out_path, "w") as f:
        f.write(f"{len(frames_joint)}\n")  # numFrame
        for fr in frames_joint:
            nbody = min(fr.get("numBody", 0), MAX_PERSON)   # NEW: MAX_PERSON 적용
            f.write(f"{nbody}\n")
            if nbody == 0:
                continue
            # ---- bodyInfo 10 fields
            bodyID = 1; clipedEdges=0; hLC=1; hLS=2; hRC=1; hRS=2; isR=0; leanX=0; leanY=0; tS=2
            f.write(f"{bodyID} {clipedEdges} {hLC} {hLS} {hRC} {hRS} {isR} {leanX} {leanY} {tS}\n")
            # ---- numJoint
            f.write("25\n")
            J = fr["J"]  # (25,3)
            for j in range(25):
                x, y, score = float(J[j,0]), float(J[j,1]), float(J[j,2])
                z = 0.0
                depthX, depthY = x, y
                colorX, colorY = x, y
                oW, oX, oY, oZ = 1.0, 0.0, 0.0, 0.0
                tState = 2.0 if score >= VIS_THRES else 1.0
                f.write(f"{x} {y} {z} {depthX} {depthY} {colorX} {colorY} {oW} {oX} {oY} {oZ} {tState}\n")

# --------- FPS 리샘플/길이 맞춤 유틸 ---------
def resample_by_fps(frames, src_fps, dst_fps):
    """원본 프레임 리스트를 dst_fps에 맞게 인덱스 리샘플"""
    if src_fps is None or src_fps <= 0:  # CHANGED: fps 불명 시 그대로
        return frames
    n = len(frames)
    if n == 0: return frames
    dur = n / float(src_fps)
    n_dst = max(1, int(round(dur * float(dst_fps))))
    idx = np.linspace(0, n-1, num=n_dst).round().astype(int)
    idx = np.clip(idx, 0, n-1)
    return [frames[i] for i in idx]

def fit_to_length(frames, T):
    """길이가 T가 되도록 잘라내기/보간/패딩(0-body)"""
    n = len(frames)
    if n == 0:
        return [{"numBody": 0} for _ in range(T)]
    if n == T:
        return frames
    if n > T:
        # 균등 샘플링으로 T개 선택
        idx = np.linspace(0, n-1, num=T).round().astype(int)
        return [frames[i] for i in idx]
    # pad 0-body
    pad = [{"numBody": 0} for _ in range(T - n)]
    return frames + pad

# --------- 비디오 -> frames_joint ---------
mp_pose = mp.solutions.pose

def video_to_skeleton_frames(video_path):
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        return [], None  # CHANGED: fps도 반환

    src_fps = cap.get(cv2.CAP_PROP_FPS) or 0.0  # CHANGED
    frames = []
    with mp_pose.Pose(static_image_mode=False, model_complexity=1,
                      enable_segmentation=False,
                      min_detection_confidence=0.5,
                      min_tracking_confidence=0.5) as pose:
        n = 0
        while True:
            ok, frame = cap.read()
            if not ok: break
            n += 1
            if MAX_FRAMES and n > MAX_FRAMES: break

            rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            res = pose.process(rgb)

            if res.pose_landmarks:
                J = mp_to_ntu25(res.pose_landmarks.landmark)  # (25,3)
                mean_vis = float(J[:,2].mean())
                if mean_vis >= 0.05:
                    frames.append({"numBody": 1, "J": J})
                else:
                    frames.append({"numBody": 0})
            else:
                frames.append({"numBody": 0})
    cap.release()
    return frames, float(src_fps)

    S = DEFAULT_SETUP
    C = DEFAULT_CAMERA
    P = DEFAULT_SUBJECT
    R = DEFAULT_REPEAT

    base = vpath.stem
    out_name = f"S{S:03d}C{C:03d}P{P:03d}R{R:03d}A{A:03d}_{base}.skeleton"
    out_path = str(Path(OUT_SKELETON)/out_name)

    frames, src_fps = video_to_skeleton_frames(str(vpath))
    if len(frames) == 0:
        # 빈 프레임이면 numFrame=0 파일 생성 (원하면 skip로 바꿔도 됨)
        with open(out_path, "w") as f: f.write("0\n")
        continue

    # NEW: 타깃 FPS로 리샘플 → 타깃 길이(TARGET_T)로 맞춤
    frames = resample_by_fps(frames, src_fps, TARGET_FPS)
    frames = fit_to_length(frames, TARGET_T)

    write_ntu_skeleton(out_path, frames)

print(f"Done. Saved .skeleton files to: {OUT_SKELETON}")
print("샘플 파일 예시:", sorted(os.listdir(OUT_SKELETON))[:5])

from pathlib import Path

ROUND_ND = None        # 예: 5 -> 소수 5자리 반올림 후 비교(근사 일치). 정확 비교는 None
FILTER_PATTERN = "*.npy"  # 예) "*_data_joint.npy" 로 좁힐 수 있음
DRY_RUN = False         # 먼저 예행연습 후, 실제 삭제하려면 False 로

def array_hash(npy_path: Path, nd=None):
    x = np.load(npy_path, mmap_mode="r")
    if nd is not None:
        a = np.round(np.asarray(x), nd)  # dtype 무관하게 반올림 적용
        a = np.ascontiguousarray(a)
    else:
        # nd=None이면 정확 동일성(바이트) 비교
        a = np.ascontiguousarray(np.asarray(x))
    return hashlib.sha256(a.tobytes()).hexdigest()[:16]

def is_flip_name(p: Path) -> bool:
    # 파일명에 "_flip" 토큰이 있으면 플립으로 간주
    return "_flip" in p.name

def label_path_of(npy_path: Path) -> Path:
    # *_data_joint.npy -> *_label.pkl 로 매핑(필요 없으면 수정/삭제)
    name = npy_path.name
    if name.endswith("_data_joint.npy"):
        return npy_path.with_name(name.replace("_data_joint.npy", "_label.pkl"))
    # 일반 규칙이 아니면 None 반환
    return npy_path.with_suffix(".pkl")  # 혹시 다른 규칙이면 적절히 수정

# 1) 해시 그룹 만들기
groups = {}
candidates = sorted(ROOT.rglob(FILTER_PATTERN))
for npy in candidates:
    try:
        h = array_hash(npy, nd=ROUND_ND)
    except Exception as e:
        print(f"[WARN] 해시 실패: {npy} -> {e}")
        continue
    groups.setdefault(h, []).append(npy)

dup = {h:paths for h,paths in groups.items() if len(paths) > 1}

print(f"[INFO] 값기반 중복 그룹 수 (nd={ROUND_ND}): {len(dup)}")

# 2) 중복 그룹 처리 정책:
#   - 플립이 아닌 파일을 우선 보존
#   - 동일 조건이면 수정시각이 가장 오래된(작은 mtime) 파일 보존
#   - 나머지는 삭제(라벨 .pkl 있으면 같이 삭제)
delete_targets = []
keep_map = {}  # h -> kept path

for h, paths in dup.items():
    # 정렬: (플립 여부, mtime)
    # 플립이 아닌 것이 우선 -> is_flip_name=False(0)이 True(1)보다 앞
    paths_sorted = sorted(paths, key=lambda p: (is_flip_name(p), p.stat().st_mtime))
    keep = paths_sorted[0]
    keep_map[h] = keep
    victims = paths_sorted[1:]

    print(f"\n[GROUP {h}] keep -> {keep}")
    for v in victims:
        delete_targets.append(v)
        print("  del ->", v)

# 3) 실제 삭제 수행(+ 라벨 동반 삭제)
def safe_unlink(p: Path):
    try:
        if not DRY_RUN and p.exists():
            p.unlink()
        return True, None
    except Exception as e:
        return False, e

deleted, failed = 0, 0
for v in delete_targets:
    ok, err = safe_unlink(v)
    if ok:
        print("[DEL] ", v)
        deleted += (0 if DRY_RUN else 1)
    else:
        print("[ERR] 삭제 실패:", v, err)
        failed += 1

    # 라벨 동반 삭제 시도 (규칙과 다르면 수정하세요)
    lbl = label_path_of(v)
    if lbl and lbl.exists():
        ok, err = safe_unlink(lbl)
        if ok:
            print("       [DEL label] ", lbl)
        else:
            print("       [ERR] 라벨 삭제 실패:", lbl, err)

print(f"\n[DONE] 삭제 완료. deleted={deleted} failed={failed} (DRY_RUN={DRY_RUN})")

# -------------------- 유틸: prefix 복원 & 라벨 탐색 --------------------
_SUFFIX_DATA_JOINT_RX = re.compile(r'(?:_data_joint(?:_[A-Za-z0-9]+)?)$')
_COPY_RX              = re.compile(r'(?:_copy\d+)$')
_AUG_RX               = re.compile(r'(?:_aug_[^_]+)$')
_SINGLE_AUG_RX        = re.compile(r'(?:_(?:ypr|yaw|pitch|roll|flip))$')  # 필요시 확장

def normalize_prefix_from_joint_name(jpath: Path) -> str:
    """
    예) S001..._squat7_ypr_copy08_data_joint_AGCN.npy  ->  S001..._squat7
    제거 순서: _data_joint(_TAG)? -> _copyNN -> _aug_* -> _(ypr|yaw|pitch|roll|flip)
    """
    s = jpath.stem
    s = _SUFFIX_DATA_JOINT_RX.sub('', s)
    s = _COPY_RX.sub('', s)
    s = _AUG_RX.sub('', s)
    s = _SINGLE_AUG_RX.sub('', s)
    return s

def find_label_for_joint(jpath: Path, debug=False):
    """
    - aug_dataset_new 안의 joint 파일에 대해:
      1) 같은 폴더에서 prefix_label*.pkl 찾기
      2) MIRROR_LABEL_ROOT/<동일한 하위폴더>/ 에서 prefix_label*.pkl 찾기  ← ★추가
      3) EXTRA_LABEL_DIRS 에서도 찾기
      4) direct replace (_data_joint_→_label_) 도 한 번 시도
    """
    prefix = normalize_prefix_from_joint_name(jpath)
    d = jpath.parent
    candidates = []

    # 0) direct replace (호환성)
    direct = Path(str(jpath).replace("_data_joint_", "_label_")).with_suffix(".pkl")
    candidates.append(direct)

    # 1) same dir
    candidates.append(d / f"{prefix}_label.pkl")
    candidates += sorted(d.glob(f"{prefix}_label*.pkl"))

    # 2) mirror dir (aug_dataset_new 하위 경로를 npy_data 로 치환)
    try:
        rel = jpath.relative_to(SRC_ROOT).parent  # 예) dumbbell_squat/
        mirror_dir = MIRROR_LABEL_ROOT / rel
        candidates.append(mirror_dir / f"{prefix}_label.pkl")
        candidates += sorted(mirror_dir.glob(f"{prefix}_label*.pkl"))
    except Exception:
        pass  # relative_to 실패 시 패스

    # 3) extra dirs
    for extra in EXTRA_LABEL_DIRS:
        extra = Path(extra)
        candidates.append(extra / f"{prefix}_label.pkl")
        candidates += sorted(extra.glob(f"{prefix}_label*.pkl"))

    # 유일화
    seen, uniq = set(), []
    for p in candidates:
        s = str(p)
        if s in seen:
            continue
        seen.add(s); uniq.append(p)

    for p in uniq:
        if p.exists():
            if debug:
                print(f"[LABEL] {jpath.name} -> {p}")
            return p

    if debug:
        print(f"[DEBUG] label candidates (first 10) for {jpath.name}:")
        for c in uniq[:10]:
            print("   -", c)
    return None

def robust_load_label(path: Path):
    p = Path(path)
    # 1) pickle
    try:
        with open(p, "rb") as f:
            return pickle.load(f)
    except Exception:
        pass
    # 2) json
    try:
        with open(p, "r", encoding="utf-8-sig") as f:
            return json.load(f)
    except Exception:
        pass
    # 3) txt (줄당 int)
    try:
        vals = []
        with open(p, "r", encoding="utf-8-sig") as f:
            for ln in f:
                t = ln.strip()
                if t:
                    try: vals.append(int(t.split()[0]))
                    except: pass
        if vals:
            return {"label_index": vals}
    except Exception:
        pass
    return None

def get_y(L, k: int, K: int):
    if L is None: return None
    # dict
    if isinstance(L, dict):
        if "label_index" in L:
            li = L["label_index"]
            if isinstance(li, (list, tuple, np.ndarray)) and len(li)>0:
                try: return int(li[min(k, len(li)-1)])
                except: return None
        if "label" in L:
            lv = L["label"]
            v = lv[min(k, len(lv)-1)] if isinstance(lv, (list, tuple, np.ndarray)) and len(lv)>0 else lv
            try: return int(v)
            except:
                try: return int(str(v))
                except: return None
        return None
    # tuple
    if isinstance(L, tuple) and len(L)==2:
        _, lab = L
        if isinstance(lab, (list, tuple, np.ndarray)) and len(lab)>0:
            try: return int(lab[min(k, len(lab)-1)])
            except: return None
        try: return int(lab)
        except: return None
    # list/ndarray
    if isinstance(L, (list, tuple, np.ndarray)) and len(L)>0:
        try: return int(L[min(k, len(L)-1)])
        except: return None
    # scalar
    if isinstance(L, (int, np.integer, str)):
        try: return int(L)
        except: return None
    return None

def single_label_dict(L, k: int, K: int) -> dict:
    if isinstance(L, dict):
        out = {}
        if "label_index" in L:
            li = L["label_index"]
            if isinstance(li, (list, tuple, np.ndarray)) and len(li)>0:
                out["label_index"] = [int(li[min(k, len(li)-1)])]
        if "label" in L:
            lv = L["label"]
            if isinstance(lv, (list, tuple, np.ndarray)) and len(lv)>0:
                out["label"] = [lv[min(k, len(lv)-1)]]
            else:
                out["label"] = [lv]
        if out: return out
    vy = get_y(L, k, K)
    return {"label_index": [int(vy) if vy is not None else 0]}

def to_NCTVM_minibatch(X, expect_v=EXPECT_V_SET, force_M=FORCE_M):
    """
    X: np.ndarray
      - (N,C,T,V,M) 또는 (C,T,V,M) 또는 기타 혼선
    반환: (K, 3, T, V, M>=force_M)
      - 여기서 K는 파일 내부 샘플 수(=N), 저장은 슬라이스별 (1,3,T,V,M)
    """
    arr = np.asarray(X)
    # 1) 이미 5D이면 (N,C,T,V,M) 가정
    if arr.ndim == 5:
        N,C,T,V,M = arr.shape
        if C != 3:
            # 채널이 앞이 아닐 수 있으니 재정렬 시도
            # C=3 축을 찾아 (N,C,T,V,M) 재배치
            shape = arr.shape
            if 3 not in shape:
                raise ValueError(f"C=3 axis not found: {shape}")
            c_ax = list(range(arr.ndim))[list(shape).index(3)]
            # 남은 축 중 가장 큰 걸 T, V는 expect 우선
            other = [i for i in range(5) if i != c_ax]
            t_ax = max(other, key=lambda i: shape[i])
            v_ax = None
            for i in other:
                if i == t_ax: continue
                if shape[i] in expect_v:
                    v_ax = i; break
            if v_ax is None:
                v_ax = [i for i in other if i != t_ax][0]
            nm_axes = [i for i in range(5) if i not in (c_ax,t_ax,v_ax)]
            # nm_axes를 N,M으로
            perm = [nm_axes[0], c_ax, t_ax, v_ax, nm_axes[-1]]
            arr = np.transpose(arr, perm)
            N,C,T,V,M = arr.shape
    # 2) 4D면 (C,T,V,M) 가정 → N=1 추가
    elif arr.ndim == 4:
        C,T,V,M = arr.shape
        if C != 3:
            # C=3 축 찾아 (C,T,V,M)로
            shape = arr.shape
            if 3 not in shape:
                raise ValueError(f"C=3 axis not found: {shape}")
            c_ax = list(range(4))[list(shape).index(3)]
            other = [i for i in range(4) if i != c_ax]
            t_ax = max(other, key=lambda i: shape[i])
            v_ax = None
            for i in other:
                if i == t_ax: continue
                if shape[i] in expect_v:
                    v_ax = i; break
            if v_ax is None:
                v_ax = [i for i in other if i != t_ax][0]
            m_ax = [i for i in range(4) if i not in (c_ax,t_ax,v_ax)][0]
            arr = np.transpose(arr, (c_ax, t_ax, v_ax, m_ax))
            C,T,V,M = arr.shape
        arr = arr[None, ...]  # N=1
        N,C,T,V,M = arr.shape
    else:
        # 그 외는 squeeze 후 재추정
        arr = np.squeeze(arr)
        if arr.ndim < 3:
            raise ValueError(f"Unsupported shape after squeeze: {arr.shape}")
        # C=3 축 찾기 -> (C,T,V,[N?],[M?]) 정렬 후 (N,C,T,V,M)
        shape = arr.shape
        if 3 not in shape:
            raise ValueError(f"C=3 axis not found: {shape}")
        c_ax = list(range(arr.ndim))[list(shape).index(3)]
        other = [i for i in range(arr.ndim) if i != c_ax]
        t_ax = max(other, key=lambda i: shape[i])
        v_ax = None
        for i in other:
            if i == t_ax: continue
            if shape[i] in expect_v:
                v_ax = i; break
        if v_ax is None:
            cand = [i for i in other if i != t_ax]
            v_ax = cand[0]
        nm_axes = [i for i in range(arr.ndim) if i not in (c_ax,t_ax,v_ax)]
        perm = [c_ax, t_ax, v_ax] + nm_axes
        arr = np.transpose(arr, perm)
        C,T,V,*nm = arr.shape
        if len(nm) == 0: N,M = 1,1
        elif len(nm) == 1: N,M = nm[0],1
        else:
            N, M = nm[0], nm[-1]
        arr = arr.reshape(C,T,V,N,M).transpose(3,0,1,2,4)
        N,C,T,V,M = arr.shape

    # V 검사
    if V not in EXPECT_V_SET:
        raise ValueError(f"Unexpected V={V} in array {arr.shape}")

    # M padding
    if M < force_M:
        pad = np.zeros((N, C, T, V, force_M - M), dtype=arr.dtype)
        arr = np.concatenate([arr, pad], axis=-1)

    return arr.astype(np.float32, copy=False)  # (N,3,T,V,>=M)

def crop_or_pad_T(x, out_T=300):
    C,T,V,M = x.shape
    if T==out_T: return x
    if T>out_T:
        start = np.random.randint(0, T-out_T+1); return x[:, start:start+out_T]
    y = np.zeros((C,out_T,V,M), dtype=x.dtype)
    start = np.random.randint(0, out_T-T+1); y[:, start:start+T] = x; return y

def temporal_shift(x, max_shift=6):
    C,T,V,M = x.shape; s = np.random.randint(-max_shift, max_shift+1)
    y = np.zeros_like(x)
    if s>0: y[:, s:] = x[:, :T-s]
    elif s<0: y[:, :T+s] = x[:, -s:]
    else: y[:] = x
    return y

def temporal_speed_nearest(x, rate_min=0.95, rate_max=1.05, out_T=300):
    C,T,V,M = x.shape; rate = np.random.uniform(rate_min, rate_max)
    L = max(1, int(round(T*rate)))
    idx = np.clip(np.round(np.linspace(0, T-1, num=L)).astype(int), 0, T-1)
    y = x[:, idx, :, :]
    if y.shape[1] < out_T:
        z = np.zeros((C,out_T,V,M), dtype=y.dtype); z[:, :y.shape[1]] = y; y = z
    elif y.shape[1] > out_T:
        y = y[:, :out_T]
    return y

def rotate_2d(x, max_deg=8):
    th = np.deg2rad(np.random.uniform(-max_deg, max_deg))
    R = np.array([[np.cos(th), -np.sin(th)],
                  [np.sin(th),  np.cos(th)]], dtype=x.dtype)
    y = x.copy(); xy = y[:2].reshape(2, -1)
    y[:2] = (R @ xy).reshape(2, x.shape[1], x.shape[2], x.shape[3]); return y

def flip_x(x):
    y = x.copy(); y[0] = -y[0]; return y

def scale_xy(x, smin=0.97, smax=1.03):
    s = np.random.uniform(smin, smax); y = x.copy(); y[:2] *= s; return y

def gaussian_jitter(x, sigma=0.003):
    return x + np.random.normal(0, sigma, size=x.shape).astype(x.dtype)

def joint_dropout(x, p=0.03):
    y = x.copy(); mask = (np.random.rand(*x.shape) < p); y[mask] = 0.0; return y

def strong_augment(x, out_T=300):
    x = crop_or_pad_T(x, out_T=out_T)
    if np.random.rand()<0.7: x = temporal_shift(x, 6)
    if np.random.rand()<0.5: x = temporal_speed_nearest(x, 0.95, 1.05, out_T=out_T)
    if np.random.rand()<0.5: x = rotate_2d(x, 8)
    if np.random.rand()<0.5: x = flip_x(x)
    if np.random.rand()<0.5: x = scale_xy(x, 0.97, 1.03)
    if np.random.rand()<0.7: x = gaussian_jitter(x, 0.003)
    if np.random.rand()<0.2: x = joint_dropout(x, 0.03)
    return x

    return x.astype(np.float32, copy=False)

# ==========================
# MP4 -> NTU .skeleton writer (with FPS/T window control)
# ==========================
#!pip install mediapipe opencv-python tqdm

import os, cv2, math, numpy as np
from pathlib import Path
from tqdm import tqdm
import mediapipe as mp

# --------- 사용자 설정 ---------
ROOT_VIDEOS  = "/content/drive/MyDrive/single_npy"     # 클래스/영상.mp4 구조 권장
OUT_SKELETON = "/content/drive/MyDrive/single_npy/skeleton"      # .skeleton 저장 폴더
os.makedirs(OUT_SKELETON, exist_ok=True)

# 파일명 메타 기본값(원본 NTU 네이밍 형식: S###C###P###R###A###.skeleton)
DEFAULT_SETUP   = 1   # S
DEFAULT_CAMERA  = 2   # C (xview에서 2,3 카메라가 train에 들어감)
DEFAULT_SUBJECT = 1   # P
DEFAULT_REPEAT  = 1   # R

# 감지/프레임 설정
VIS_THRES  = 0.5
MAX_FRAMES = None        # None이면 전체, 숫자면 그만큼만(디버그)

# --------- 추가: 타깃 FPS / 길이 / 인원 ---------
TARGET_FPS = 30          # NEW: NTU는 30fps 기준. 원본 fps 상관없이 이 기준으로 근사 리샘플
TARGET_T   = 300         # NEW: 최종 프레임 길이(윈도우) 고정. 실시간/파인튜닝이면 30~60 권장
MAX_PERSON = 1           # NEW: 최대 인원. (MediaPipe Pose는 1인이라 현재 0/1만 나옴)

# --------- MediaPipe → NTU25 매핑 ---------
def mp_to_ntu25(landmarks):
    """ landmarks: list of 33 normalized (x,y,visibility) in [0,1]
        return (25,3) with [x,y,score] in [0,1]
    """
    def get(i):
        lm = landmarks[i]
        return np.array([lm.x, lm.y, getattr(lm, "visibility", 1.0)], dtype=np.float32)
    def mid(a,b): return (a+b)/2.0

    # key indices
    LSH, RSH = get(11), get(12)
    LHP, RHP = get(23), get(24)
    L_EL, R_EL = get(13), get(14)
    L_WR, R_WR = get(15), get(16)
    L_ID, R_ID = get(19), get(20)
    L_TB, R_TB = get(21), get(22)
    L_KN, R_KN = get(25), get(26)
    L_AN, R_AN = get(27), get(28)
    L_FI, R_FI = get(31), get(32)
    SHO = mid(LSH, RSH)  # spine shoulder
    HIP = mid(LHP, RHP)  # spine base
    NEK = SHO            # neck approx
    HED = get(0)         # head ~ nose

    J = np.stack([
        HIP,           # 0 spine base
        (HIP+SHO)/2.0, # 1 spine mid (approx)
        NEK,           # 2 neck
        HED,           # 3 head
        LSH, L_EL, L_WR, L_ID,       # 4..7
        RSH, R_EL, R_WR, R_ID,       # 8..11
        LHP, L_KN, L_AN, L_FI,       # 12..15
        RHP, R_KN, R_AN, R_FI,       # 16..19
        SHO, L_ID, L_TB, R_ID, R_TB  # 20..24
    ], axis=0).astype(np.float32)

    J[:, :2] = np.clip(J[:, :2], 0.0, 1.0)
    J[:, 2]  = np.clip(J[:, 2], 0.0, 1.0)  # score
    return J  # (25,3)

# --------- .skeleton 파일 작성기 ---------
def write_ntu_skeleton(out_path, frames_joint):
    """
    frames_joint: list of dict per frame:
      - if no body: {'numBody': 0}
      - if body:    {'numBody': 1, 'J': (25,3) normalized}
    Writes NTU skeleton text format compatible with read_skeleton_filter.
    """
    with open(out_path, "w") as f:
        f.write(f"{len(frames_joint)}\n")  # numFrame
        for fr in frames_joint:
            nbody = min(fr.get("numBody", 0), MAX_PERSON)   # NEW: MAX_PERSON 적용
            f.write(f"{nbody}\n")
            if nbody == 0:
                continue
            # ---- bodyInfo 10 fields
            bodyID = 1; clipedEdges=0; hLC=1; hLS=2; hRC=1; hRS=2; isR=0; leanX=0; leanY=0; tS=2
            f.write(f"{bodyID} {clipedEdges} {hLC} {hLS} {hRC} {hRS} {isR} {leanX} {leanY} {tS}\n")
            # ---- numJoint
            f.write("25\n")
            J = fr["J"]  # (25,3)
            for j in range(25):
                x, y, score = float(J[j,0]), float(J[j,1]), float(J[j,2])
                z = 0.0
                depthX, depthY = x, y
                colorX, colorY = x, y
                oW, oX, oY, oZ = 1.0, 0.0, 0.0, 0.0
                tState = 2.0 if score >= VIS_THRES else 1.0
                f.write(f"{x} {y} {z} {depthX} {depthY} {colorX} {colorY} {oW} {oX} {oY} {oZ} {tState}\n")

# --------- FPS 리샘플/길이 맞춤 유틸 ---------
def resample_by_fps(frames, src_fps, dst_fps):
    """원본 프레임 리스트를 dst_fps에 맞게 인덱스 리샘플"""
    if src_fps is None or src_fps <= 0:  # CHANGED: fps 불명 시 그대로
        return frames
    n = len(frames)
    if n == 0: return frames
    dur = n / float(src_fps)
    n_dst = max(1, int(round(dur * float(dst_fps))))
    idx = np.linspace(0, n-1, num=n_dst).round().astype(int)
    idx = np.clip(idx, 0, n-1)
    return [frames[i] for i in idx]

def fit_to_length(frames, T):
    """길이가 T가 되도록 잘라내기/보간/패딩(0-body)"""
    n = len(frames)
    if n == 0:
        return [{"numBody": 0} for _ in range(T)]
    if n == T:
        return frames
    if n > T:
        # 균등 샘플링으로 T개 선택
        idx = np.linspace(0, n-1, num=T).round().astype(int)
        return [frames[i] for i in idx]
    # pad 0-body
    pad = [{"numBody": 0} for _ in range(T - n)]
    return frames + pad

# --------- 비디오 -> frames_joint ---------
mp_pose = mp.solutions.pose

def video_to_skeleton_frames(video_path):
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        return [], None  # CHANGED: fps도 반환

    src_fps = cap.get(cv2.CAP_PROP_FPS) or 0.0  # CHANGED
    frames = []
    with mp_pose.Pose(static_image_mode=False, model_complexity=1,
                      enable_segmentation=False,
                      min_detection_confidence=0.5,
                      min_tracking_confidence=0.5) as pose:
        n = 0
        while True:
            ok, frame = cap.read()
            if not ok: break
            n += 1
            if MAX_FRAMES and n > MAX_FRAMES: break

            rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            res = pose.process(rgb)

            if res.pose_landmarks:
                J = mp_to_ntu25(res.pose_landmarks.landmark)  # (25,3)
                mean_vis = float(J[:,2].mean())
                if mean_vis >= 0.05:
                    frames.append({"numBody": 1, "J": J})
                else:
                    frames.append({"numBody": 0})
            else:
                frames.append({"numBody": 0})
    cap.release()
    return frames, float(src_fps)

# --------- 클래스→AID 매핑 만들기 ---------
video_paths = sorted([p for p in Path(ROOT_VIDEOS).rglob("*.mp4")])
if not video_paths:
    raise SystemExit(f"No mp4 files under {ROOT_VIDEOS}")

classes = sorted({p.parent.name for p in video_paths})
cls2aid = {c:i+1 for i,c in enumerate(classes)}  # 1..K
print("Class -> AID mapping:", cls2aid)

# --------- 변환 실행 ---------
for vpath in tqdm(video_paths, desc="Converting"):
    cls = vpath.parent.name
    A = cls2aid[cls]           # action id 1..K
    S = DEFAULT_SETUP
    C = DEFAULT_CAMERA
    P = DEFAULT_SUBJECT
    R = DEFAULT_REPEAT

    base = vpath.stem
    out_name = f"S{S:03d}C{C:03d}P{P:03d}R{R:03d}A{A:03d}_{base}.skeleton"
    out_path = str(Path(OUT_SKELETON)/out_name)

    frames, src_fps = video_to_skeleton_frames(str(vpath))
    if len(frames) == 0:
        # 빈 프레임이면 numFrame=0 파일 생성 (원하면 skip로 바꿔도 됨)
        with open(out_path, "w") as f: f.write("0\n")
        continue

    # NEW: 타깃 FPS로 리샘플 → 타깃 길이(TARGET_T)로 맞춤
    frames = resample_by_fps(frames, src_fps, TARGET_FPS)
    frames = fit_to_length(frames, TARGET_T)

    write_ntu_skeleton(out_path, frames)

print(f"Done. Saved .skeleton files to: {OUT_SKELETON}")
print("샘플 파일 예시:", sorted(os.listdir(OUT_SKELETON))[:5])
